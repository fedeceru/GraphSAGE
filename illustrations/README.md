# Illustrations

Static, purely didactic images used in `notebook.ipynb`'s theoretical sections
(§3.1 and §6.1). Unlike everything under `results/` (which is real output from
the actual PPI reproduction pipeline — real training runs, real metrics), the
two images here are conceptual diagrams on synthetic, made-up data, generated
once and committed as plain images. They are **not** regenerated by any script
in this repo, and are kept in this separate folder specifically so that
distinction is unambiguous: nothing under `results/` should ever be a static
asset, and nothing here should ever be mistaken for a measured result.

- **`pipeline_illustration.png`** — the three-step Sample → Aggregate →
  Predict pipeline (Algorithm 1's forward pass plus the unsupervised
  objective), drawn on a synthetic 18-node graph. Referenced in notebook.ipynb
  §3.1.
- **`aggregator_architectures.png`** — the four AGGREGATE_k operations (mean,
  GCN, max-pooling, LSTM) compared side by side as small node-and-edge
  diagrams. Referenced in notebook.ipynb §6.1.

## Regenerating these images

The matplotlib code that produced them has been deliberately removed from
`notebook.ipynb` (it has no dependency on any PPI data or trained model, so
it doesn't belong in the reproduction pipeline) and is preserved here instead,
for the rare case one of these diagrams needs to be redrawn or restyled.
Running either snippet from the repo root (with the project's `.venv`
active) regenerates the corresponding file in this folder.

### `pipeline_illustration.png`

```python
import os
import numpy as np
import networkx as nx
import matplotlib.pyplot as plt
from matplotlib.lines import Line2D
from matplotlib.patches import FancyArrowPatch

rng = np.random.RandomState(7)

# a small synthetic 4-regular graph, purely for illustration (this is NOT PPI data)
demo_graph = nx.random_regular_graph(d=4, n=18, seed=7)
target_node = 0
S1, S2 = 3, 2

one_hop_all = list(demo_graph.neighbors(target_node))
one_hop_sampled = list(rng.choice(one_hop_all, size=min(S1, len(one_hop_all)), replace=False))

two_hop_sampled = set()
for u in one_hop_sampled:
    candidates = [n for n in demo_graph.neighbors(u) if n != target_node and n not in one_hop_sampled]
    if candidates:
        picked = rng.choice(candidates, size=min(S2, len(candidates)), replace=False)
        two_hop_sampled.update(picked)

color_target, color_1hop, color_2hop, color_unsampled = "#2a78d6", "#1baf7a", "#eda100", "#e1e0d9"
node_colors = []
for n in demo_graph.nodes():
    if n == target_node:
        node_colors.append(color_target)
    elif n in one_hop_sampled:
        node_colors.append(color_1hop)
    elif n in two_hop_sampled:
        node_colors.append(color_2hop)
    else:
        node_colors.append(color_unsampled)

np.random.seed(7)  # networkx 1.11's spring_layout has no `seed` kwarg; seed the global RNG instead
pos = nx.spring_layout(demo_graph)
xs = [pos[n][0] for n in demo_graph.nodes()]
ys = [pos[n][1] for n in demo_graph.nodes()]


def draw_base_edges(ax):
    for u, v in demo_graph.edges():
        x0, y0 = pos[u]
        x1, y1 = pos[v]
        ax.plot([x0, x1], [y0, y1], color="#c3c2b7", linewidth=1, zorder=1)


# Draw with plain matplotlib (rather than nx.draw_*) to avoid a networkx
# 1.11 / matplotlib incompatibility in networkx's own drawing helpers, and
# for full control over the mark style. All three panels are combined into
# one sheet (rather than three separate figures) so the pipeline reads as a
# single connected story, cf. the paper's own Figure 1.
fig, axes = plt.subplots(1, 3, figsize=(19, 7.5))

# ---------------------------------------------------------------- Panel 1: sampling
ax = axes[0]
draw_base_edges(ax)
for u in one_hop_sampled:
    x0, y0 = pos[target_node]
    x1, y1 = pos[u]
    ax.annotate("", xy=(x1, y1), xytext=(x0, y0),
                arrowprops=dict(arrowstyle="-|>", color=color_1hop, lw=2.2,
                                shrinkA=14, shrinkB=12, alpha=0.9), zorder=2)
for u in one_hop_sampled:
    for w in demo_graph.neighbors(u):
        if w in two_hop_sampled:
            x0, y0 = pos[u]
            x1, y1 = pos[w]
            ax.annotate("", xy=(x1, y1), xytext=(x0, y0),
                        arrowprops=dict(arrowstyle="-|>", color=color_2hop, lw=1.8,
                                        shrinkA=12, shrinkB=10, alpha=0.85), zorder=2)
ax.scatter(xs, ys, s=260, c=node_colors, edgecolors="#0b0b0b", linewidths=0.7, zorder=3)
ax.axis("off")
ax.set_aspect("equal")
ax.legend(handles=[
    Line2D([0], [0], marker="o", color="w", markerfacecolor=color_target, markersize=11, label="Target $v$"),
    Line2D([0], [0], marker="o", color="w", markerfacecolor=color_1hop, markersize=11, label="Sampled 1-hop ($S_1$)"),
    Line2D([0], [0], marker="o", color="w", markerfacecolor=color_2hop, markersize=11, label="Sampled 2-hop ($S_2$)"),
    Line2D([0], [0], marker="o", color="w", markerfacecolor=color_unsampled, markersize=11, label="Not sampled"),
], loc="upper left", frameon=False, fontsize=8, handletextpad=0.3, labelspacing=0.4)
ax.set_title("1. Sample neighborhood", y=-0.08, fontsize=13, fontweight="bold")

# ---------------------------------------------------------------- Panel 2: aggregation
ax = axes[1]
draw_base_edges(ax)
for u in one_hop_sampled:
    for w in demo_graph.neighbors(u):
        if w in two_hop_sampled:
            x0, y0 = pos[w]
            x1, y1 = pos[u]
            ax.annotate("", xy=(x1, y1), xytext=(x0, y0),
                        arrowprops=dict(arrowstyle="-|>", color=color_2hop, lw=2.2,
                                        shrinkA=10, shrinkB=12, alpha=0.9), zorder=2)
for u in one_hop_sampled:
    x0, y0 = pos[u]
    x1, y1 = pos[target_node]
    ax.annotate("", xy=(x1, y1), xytext=(x0, y0),
                arrowprops=dict(arrowstyle="-|>", color=color_1hop, lw=2.8,
                                shrinkA=10, shrinkB=15, alpha=0.95), zorder=2)
ax.scatter(xs, ys, s=260, c=node_colors, edgecolors="#0b0b0b", linewidths=0.7, zorder=3)
ax.axis("off")
ax.set_aspect("equal")
ax.legend(handles=[
    Line2D([0], [0], color=color_2hop, lw=2.2, label="$k{=}1$: 2-hop $\\to$ 1-hop"),
    Line2D([0], [0], color=color_1hop, lw=2.8, label="$k{=}2$: 1-hop $\\to$ target"),
], loc="upper left", frameon=False, fontsize=8, handletextpad=0.3, labelspacing=0.4)
ax.set_title("2. Aggregate feature information\nfrom neighbors", y=-0.14, fontsize=13, fontweight="bold")

# ---------------------------------------------------------------- Panel 3: prediction
ax = axes[2]
draw_base_edges(ax)

rng_pred = np.random.RandomState(11)


def random_walk(graph, start, length, rng):
    walk = [start]
    for _ in range(length):
        neighbors = list(graph.neighbors(walk[-1]))
        if not neighbors:
            break
        walk.append(rng.choice(neighbors))
    return walk


walk = random_walk(demo_graph, target_node, length=4, rng=rng_pred)
positive_node = walk[-1] if walk[-1] != target_node else walk[-2]

candidates = [n for n in demo_graph.nodes() if n not in (target_node, positive_node)]
negative_nodes = list(rng_pred.choice(candidates, size=4, replace=False))

color_pos, color_neg = "#6b4fbb", "#e34948"

for a, b in zip(walk[:-1], walk[1:]):
    x0, y0 = pos[a]
    x1, y1 = pos[b]
    ax.plot([x0, x1], [y0, y1], color=color_pos, linewidth=2.2, linestyle="--", alpha=0.85, zorder=2)

x0, y0 = pos[target_node]
x1, y1 = pos[positive_node]
ax.annotate("", xy=(x1, y1), xytext=(x0, y0),
            arrowprops=dict(arrowstyle="<|-|>", color=color_pos, lw=2.6, alpha=0.95), zorder=3)

for v_n in negative_nodes:
    x0, y0 = pos[target_node]
    x1, y1 = pos[v_n]
    ax.plot([x0, x1], [y0, y1], color=color_neg, linewidth=1.5, linestyle=":", alpha=0.8, zorder=2)

node_colors_pred = []
for n in demo_graph.nodes():
    if n == target_node:
        node_colors_pred.append(color_target)
    elif n == positive_node:
        node_colors_pred.append(color_pos)
    elif n in negative_nodes:
        node_colors_pred.append(color_neg)
    else:
        node_colors_pred.append(color_unsampled)

ax.scatter(xs, ys, s=260, c=node_colors_pred, edgecolors="#0b0b0b", linewidths=0.7, zorder=4)
ax.axis("off")
ax.set_aspect("equal")
ax.legend(handles=[
    Line2D([0], [0], marker="o", color="w", markerfacecolor=color_target, markersize=11, label="Anchor $u$"),
    Line2D([0], [0], marker="o", color="w", markerfacecolor=color_pos, markersize=11, label="Positive $v$ (walk)"),
    Line2D([0], [0], marker="o", color="w", markerfacecolor=color_neg, markersize=11, label="Negatives $v_n$"),
], loc="upper left", frameon=False, fontsize=8, handletextpad=0.3, labelspacing=0.4)
ax.set_title("3. Predict graph context\n(unsupervised objective, Eq. 1)", y=-0.14, fontsize=13, fontweight="bold")

fig.suptitle("The GraphSAGE forward pass, step by step (cf. paper Figure 1)", fontsize=15, y=1.02)
fig.tight_layout()

# transition arrows between panels, drawn in figure coordinates once the
# final subplot positions are known -- these are the visual "cf. paper
# Figure 1" connectors between sample -> aggregate -> predict
for i in range(2):
    box_a = axes[i].get_position()
    box_b = axes[i + 1].get_position()
    y_mid = (box_a.y0 + box_a.y1) / 2
    fig.add_artist(FancyArrowPatch(
        (box_a.x1 + 0.01, y_mid), (box_b.x0 - 0.01, y_mid),
        transform=fig.transFigure, arrowstyle="-|>", mutation_scale=28,
        color="#2a2a2a", lw=2.2, zorder=10))

out_dir = os.path.join(os.getcwd(), "illustrations")
os.makedirs(out_dir, exist_ok=True)
fig.savefig(os.path.join(out_dir, "pipeline_illustration.png"), dpi=140, bbox_inches="tight")
plt.show()
```

### `aggregator_architectures.png`

```python
import os
import numpy as np
import matplotlib.pyplot as plt
from matplotlib.patches import Circle, Wedge, Rectangle, FancyArrowPatch
from matplotlib.colors import to_rgb, to_hex

MODEL_COLORS = {
    "GraphSAGE-GCN": "#eb6834",
    "GraphSAGE-mean": "#2a78d6",
    "GraphSAGE-LSTM": "#4a3aa7",
    "GraphSAGE-pool": "#1baf7a",
}

# Reuses the self/neighbor color convention from the pipeline illustration
# above (color_target/color_1hop), so "self" and "neighbor" mean the same
# thing visually everywhere. Neighbors get 3 distinct shades so a
# blended/selected/recency-weighted result is visibly different from just
# "the same green again".
color_self = "#2a78d6"
neigh_shades = ["#8fe0bd", "#1baf7a", "#0d7a4e"]  # light / base / dark green


def _blend(colors, weights=None):
    '''Numeric RGB average -- used so a "blended" node's fill color is the
    actual mix of its inputs, not just a stand-in color.'''
    rgbs = [to_rgb(c) for c in colors]
    if weights is None:
        weights = [1] * len(rgbs)
    total = sum(weights)
    r = sum(c[0] * w for c, w in zip(rgbs, weights)) / total
    g = sum(c[1] * w for c, w in zip(rgbs, weights)) / total
    b = sum(c[2] * w for c, w in zip(rgbs, weights)) / total
    return to_hex((r, g, b))


def _node(ax, xy, color, r=0.32, lw=1.1, ring=False):
    edge = "#eda100" if ring else "#0b0b0b"
    ax.add_patch(Circle(xy, r, facecolor=color, edgecolor=edge, linewidth=2.2 if ring else lw, zorder=3))


def _arrow(ax, p0, p1, color="#898781", lw=1.3, rad=0.0):
    ax.add_patch(FancyArrowPatch(p0, p1, arrowstyle="-|>", color=color, lw=lw,
                                  shrinkA=10, shrinkB=10, mutation_scale=14,
                                  connectionstyle=f"arc3,rad={rad}", zorder=2))


def _split_circle(ax, xy, r, color_left, color_right):
    '''Half-and-half circle standing in for CONCAT(self, aggregated) --
    literally two halves joined, instead of the word "CONCAT".'''
    x, y = xy
    ax.add_patch(Wedge((x, y), r, 90, 270, facecolor=color_left, edgecolor="#0b0b0b", linewidth=1.1, zorder=3))
    ax.add_patch(Wedge((x, y), r, -90, 90, facecolor=color_right, edgecolor="#0b0b0b", linewidth=1.1, zorder=3))


def _sigma_icon(ax, xy, color):
    '''A literal S-curve standing in for sigma(W*), instead of the formula.'''
    x, y = xy
    ax.add_patch(Rectangle((x - 0.48, y - 0.34), 0.96, 0.68, fill=False, edgecolor="#c9c8c2", lw=0.9, zorder=2))
    t = np.linspace(-1, 1, 60)
    curve = np.tanh(t * 2.5)
    ax.plot(x + t * 0.36, y + curve * 0.24, color=color, lw=2.0, zorder=3)


SELF_XY = (1.3, 1.1)
FAN_XY = [(1.3, 5.5), (0.75, 4.5), (1.3, 3.6)]     # mean/gcn: fanned so all 3 arrows have distinct slopes
ROW_XY = [(1.0, 4.6), (1.7, 4.6), (2.4, 4.6)]      # pool: horizontal row, needs vertical room for value bars
CHAIN_XY = [(0.6, 4.6), (1.5, 4.95), (2.4, 4.6)]   # lstm: left-to-right, order matters
STAGE2_XY = (4.0, 4.5)      # aggregated result (mean/pool/lstm keep a separate self branch)
GCN_MERGE_XY = (4.3, 2.9)   # gcn merges self+neighbors directly -- no separate branch, one fewer stage
COMBINED_XY = (6.3, 2.9)    # CONCAT split-circle (mean/pool/lstm only)
SIGMA_XY = (7.9, 2.9)
OUTPUT_XY = (9.3, 2.9)


def draw_aggregator_panel(ax, kind, title, panel_color):
    '''One small node-and-edge diagram per aggregator: neighbors -> combine
    -> CONCAT with self (skipped for GCN) -> sigma(W*) -> output. The
    *combine* step is drawn differently per kind so the operation itself is
    visible, not just labeled:
      mean -> neighbor colors blend into one node
      gcn  -> self + neighbors blend directly into one node (no CONCAT step)
      pool -> only the arg-max neighbor (gold ring) survives
      lstm -> neighbors processed in a shuffled left-to-right chain
    '''
    ax.set_xlim(0, 10)
    ax.set_ylim(0, 6)
    ax.axis("off")
    ax.set_aspect("equal")

    _node(ax, SELF_XY, color_self)

    if kind == "mean":
        for xy, c in zip(FAN_XY, neigh_shades):
            _node(ax, xy, c, r=0.26)
        for xy in FAN_XY:
            _arrow(ax, xy, STAGE2_XY)
        agg_color = _blend(neigh_shades)
        _node(ax, STAGE2_XY, agg_color)
        _arrow(ax, STAGE2_XY, COMBINED_XY)
        _arrow(ax, SELF_XY, COMBINED_XY)
        _split_circle(ax, COMBINED_XY, 0.36, color_self, agg_color)
        _arrow(ax, COMBINED_XY, SIGMA_XY)
        _sigma_icon(ax, SIGMA_XY, panel_color)
        _arrow(ax, SIGMA_XY, OUTPUT_XY)
        _node(ax, OUTPUT_XY, panel_color, r=0.38)

    elif kind == "gcn":
        for xy, c in zip(FAN_XY, neigh_shades):
            _node(ax, xy, c, r=0.26)
        for xy in FAN_XY:
            _arrow(ax, xy, GCN_MERGE_XY)
        _arrow(ax, SELF_XY, GCN_MERGE_XY)
        merge_color = _blend([color_self] + neigh_shades)
        _node(ax, GCN_MERGE_XY, merge_color, r=0.4)
        _arrow(ax, GCN_MERGE_XY, SIGMA_XY)
        _sigma_icon(ax, SIGMA_XY, panel_color)
        _arrow(ax, SIGMA_XY, OUTPUT_XY)
        _node(ax, OUTPUT_XY, panel_color, r=0.38)

    elif kind == "pool":
        bar_h = [0.42, 0.68, 0.32]  # relative post-MLP scalar per neighbor, for illustration
        max_i = int(np.argmax(bar_h))
        for i, (xy, c, h) in enumerate(zip(ROW_XY, neigh_shades, bar_h)):
            bx, by = xy
            ax.add_patch(Rectangle((bx - 0.06, by + 0.28), 0.12, h, facecolor=c, edgecolor="none", zorder=2))
            _node(ax, xy, c, r=0.26, ring=(i == max_i))
        _arrow(ax, ROW_XY[max_i], STAGE2_XY, color="#eda100", lw=2.0)
        agg_color = neigh_shades[max_i]  # max-pool selects, it doesn't blend
        _node(ax, STAGE2_XY, agg_color)
        _arrow(ax, STAGE2_XY, COMBINED_XY)
        _arrow(ax, SELF_XY, COMBINED_XY)
        _split_circle(ax, COMBINED_XY, 0.36, color_self, agg_color)
        _arrow(ax, COMBINED_XY, SIGMA_XY)
        _sigma_icon(ax, SIGMA_XY, panel_color)
        _arrow(ax, SIGMA_XY, OUTPUT_XY)
        _node(ax, OUTPUT_XY, panel_color, r=0.38)

    elif kind == "lstm":
        for xy, c in zip(CHAIN_XY, neigh_shades):
            _node(ax, xy, c, r=0.26)
        for p0, p1 in zip(CHAIN_XY[:-1], CHAIN_XY[1:]):
            _arrow(ax, p0, p1, color="#6b6b6b", lw=1.4)
        loop_a = (CHAIN_XY[0][0], CHAIN_XY[0][1] + 0.55)
        loop_b = (CHAIN_XY[2][0], CHAIN_XY[2][1] + 0.55)
        _arrow(ax, loop_a, loop_b, color="#b9b8b2", lw=1.1, rad=-0.6)
        _arrow(ax, loop_b, loop_a, color="#b9b8b2", lw=1.1, rad=-0.6)
        _arrow(ax, CHAIN_XY[-1], STAGE2_XY)
        agg_color = _blend(neigh_shades, weights=[0.15, 0.25, 0.6])  # last hidden state weighted heaviest
        _node(ax, STAGE2_XY, agg_color)
        _arrow(ax, STAGE2_XY, COMBINED_XY)
        _arrow(ax, SELF_XY, COMBINED_XY)
        _split_circle(ax, COMBINED_XY, 0.36, color_self, agg_color)
        _arrow(ax, COMBINED_XY, SIGMA_XY)
        _sigma_icon(ax, SIGMA_XY, panel_color)
        _arrow(ax, SIGMA_XY, OUTPUT_XY)
        _node(ax, OUTPUT_XY, panel_color, r=0.38)

    ax.set_title(title, fontsize=11, color=panel_color, fontweight="bold")


fig, axes = plt.subplots(2, 2, figsize=(13, 8.5))
draw_aggregator_panel(axes[0, 0], "mean", "GraphSAGE-mean (Sec. 3.3)", MODEL_COLORS["GraphSAGE-mean"])
draw_aggregator_panel(axes[0, 1], "gcn", "GraphSAGE-GCN (Eq. 2)", MODEL_COLORS["GraphSAGE-GCN"])
draw_aggregator_panel(axes[1, 0], "pool", "GraphSAGE-pool (Eq. 3)", MODEL_COLORS["GraphSAGE-pool"])
draw_aggregator_panel(axes[1, 1], "lstm", "GraphSAGE-LSTM (Sec. 3.3)", MODEL_COLORS["GraphSAGE-LSTM"])
fig.suptitle("The four aggregator architectures compared", fontsize=14)
fig.tight_layout()

out_dir = os.path.join(os.getcwd(), "illustrations")
os.makedirs(out_dir, exist_ok=True)
fig.savefig(os.path.join(out_dir, "aggregator_architectures.png"), dpi=140)
plt.show()
```
